主要用于找错误数据，来找代码bug

方法：写一个暴力的代码，一定要保证这个代码的正确性。通过随机数造数据，将这个数据给暴力代码和自己写的代码，看输出是否一致。
1，你想要测的方法a（最优解）
2，实现复杂度不好但是容易实现的方法b（暴力解）
3，实现一个随机样本产生器（长度也随机、值也随机）
4，把方法a和方法b跑相同的输入样本，看看得到的结果是否一样
5，如果有一个随机样本使得比对结果不一致，打印这个出错的样本进行人工干预，改对方法a和方法b
6，当样本数量很多时比对测试依然正确，可以确定方法a（最优解）已经正确。 



对拍代码
```c++
int main()
{
    // ---------- 第一步：编译三个源文件 ----------
    // 使用 system 函数调用命令行编译器 g++，分别编译：
    // shuju.cpp  -> 数据生成器，生成随机测试数据
    // my.cpp     -> 你的解题程序（待测试）
    // ans.cpp    -> 标准答案程序（暴力/正确解法）
    // -std=c++11 指定使用 C++11 标准编译，-o 指定输出可执行文件名
    system("g++ -std=c++11 shuju.cpp -o shuju.exe");
    system("g++ -std=c++11 my.cpp -o my.exe");
    system("g++ -std=c++11 ans.cpp -o ans.exe");

    // ---------- 第二步：进入无限对拍循环 ----------
    while(1)   // 一直运行，直到发现两组输出不同时跳出
    {
        // 1. 运行数据生成器，将随机数据输出到 shuju.txt 文件
        //    "> shuju.txt" 表示重定向标准输出到文件
        system("shuju.exe > shuju.txt");

        // 2. 用标准答案程序处理输入数据 shuju.txt，结果输出到 ans.txt
        //    "< shuju.txt" 表示从文件重定向标准输入
        system("ans.exe < shuju.txt > ans.txt");

        // 3. 用你的程序处理同样的输入，结果输出到 my.txt
        system("my.exe < shuju.txt > my.txt");

        // 4. 比较两个输出文件是否一致
        //    fc 是 Windows 下的文件比较命令，如果文件不同返回非零值，相同返回 0
        //    system 的返回值就是 fc 命令的退出码
        if(system("fc ans.txt my.txt"))   // 如果返回值非零，说明文件不同
        {
            cout << "Wrong Answer" << endl;   // 输出错误提示
            break;                            // 跳出无限循环，停止对拍
        }
        else
        {
            cout << "Ac" << endl;   // 否则输出通过（Accepted），继续下一轮测试
        }
    }

    return 0;   // 程序正常结束
}
```
