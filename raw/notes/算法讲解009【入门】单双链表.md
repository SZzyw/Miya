数据结构：链表
#### 单链表
单链表：每个节点至少有2个变量，当前节点的值是多少，下一个节点是哪个。最后一个节点连接-1，初始节点为0。
举例(下标|值|下一个节点):0|0|1 -> 1|24|5 -> 5|32|7 -> 7|63|-1
```c++
struct Node {  
    int val;  
    int next;  
    Node(int val=-1,int next=-1):val(val),next(next){}  
};
```

```c++
void init() {  
    arr.push_back(Node(0,-1));  
}
```

#### 单链表插入
单链表在某点后插入，就是将a->c变为a->b->c，我们只需要将添加点b的next设为c然后将a的next设为b就行了。
举例(跟着上一个例子):在下标1后面插入21
```c++
void insert(int x,int y) {  //在x后面插入y
    arr.push_back(Node(y,arr[x].next));  
    arr[x].next=arr.size()-1;  
}
```
#### 单链表删除
单链表将某点的后点删除，就是将a->b->c变为a->c，我们只需要将点a的next设为c就行了。
举例(跟着上一个例子):在下标1后面点删了
```c++
void erase(int x) {   //将x后面点删了
    arr[x].next=arr[arr[x].next].next;  
}
```
#### 模板题
[B3631 单向链表 - 洛谷](https://www.luogu.com.cn/problem/B3631)
```c++
#include <bits/stdc++.h>
using namespace std;
const int N = 1e6 + 10;

struct Node {
    int val;
    int next;

    Node(int val = -1, int next = -1) : val(val), next(next) {
    }
};

vector<Node> arr;
vector<int> fin(N);

void init() {
    arr.push_back(Node(0, -1));
}

void insert(int x, int y) {
    x = fin[x];
    fin[y] = arr.size();
    arr.push_back(Node(y, arr[x].next));
    arr[x].next = fin[y];
}

int coutNode(int x) {
    if (arr[fin[x]].next == -1)
        return 0;
    return arr[arr[fin[x]].next].val;
}

void erase(int x) {
    x = fin[x];
    arr[x].next = arr[arr[x].next].next;
}

void solve() {
    int q;
    cin >> q;
    init();
    insert(0, 1);
    while (q--) {
        int a, x, y;
        cin >> a;
        if (a == 1) {
            cin >> x >> y;
            insert(x, y);
        } else if (a == 2) {
            cin >> x;
            cout << coutNode(x) << endl;
        } else {
            cin >> x;
            erase(x);
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    solve();
    return 0;
}

```
#### 双链表
双链表：在单链表的基础上，增加了上一个节点是哪个。此时每个节点至少有3个变量，当前节点的值是多少，下一个节点是哪个，上一个节点是哪个。最后一个节点连接0，初始节点为0。
举例(下标|值|下一个节点|上一个节点):0|0|1|7 -><- 1|24|5|0 -><- 5|32|7|1 -><- 7|63|0|5-><-
```c++
struct Node {  
    int val;  
    int next;  
    int pre;  
    Node(int val=-1,int next=-1,int pre=-1):val(val),next(next),pre(pre){}  
};
```

#### 双链表插入
双链表在某点后插入，就是将a-><-c变为a-><-b-><-c，我们只需要将添加点b的next设为c,b的up设为a，然后将a的next设为b，c的up设为b。
举例(跟着上一个例子):在下标1后面插入21
```c++
void insert(int x,int y) {   //在x后面插入y
    int z=arr[x].next;
    arr.push_back(Node(y,x,z));
    arr[z].pre=arr.size()-1;
    arr[x].next=arr.size()-1;
}
```

#### 双链表删除
将双链表某点删了，就是将a-><-b-><-c变为a-><-c，我们只需要将a的next这位c，c的up设为b。
举例(跟着上一个例子):在值为21的点删了
```c++
void erase(int x) {
    int y=x;
    x=arr[y].pre;
    int z=arr[y].next;
    arr[x].next=z;
    arr[z].pre=x;
}
```

#### 模板题
[B4324 【模板】双向链表 - 洛谷](https://www.luogu.com.cn/problem/B4324)
```c++
#include <bits/stdc++.h>  
using namespace std;  
const int N = 1e6 + 10;  
  
struct Node {  
    int val;  
    int pre;  
    int next;  
  
    Node(int val = -1, int pre = -1, int next = -1) : val(val), pre(pre), next(next) {  
    }};  
  
vector<Node> arr;  
vector<int> fin;  
  
void init() {  
    arr.push_back(Node(0, 0, 0));  
}  
  
void insertLeft(int x, int y) {  
    if (x == y)  
        return;  
    x = fin[x];  
    x = arr[x].pre;  
    int z = arr[x].next;  
    fin[y] = arr.size();  
    arr.push_back(Node(y, x, z));  
    arr[z].pre = arr.size() - 1;  
    arr[x].next = arr.size() - 1;  
}  
  
void insert(int x, int y) {  
    if (x == y)  
        return;  
    x = fin[x];  
    int z = arr[x].next;  
    fin[y] = arr.size();  
    arr.push_back(Node(y, x, z));  
    arr[z].pre = arr.size() - 1;  
    arr[x].next = arr.size() - 1;  
}  
  
void erase(int x) {  
    if (fin[x] == -1)  
        return;  
    x = fin[x];  
    int y = x;  
    x = arr[x].pre;  
    int z = arr[y].next;  
    arr[x].next = z;  
    arr[z].pre = x;  
    fin[arr[y].val] = -1; //注意  
}  
  
void solve() {  
    int n, m;  
    cin >> n >> m;  
    fin.resize(n + 5);  
    init();  
    for (int i = 1; i <= n; i++) {  
        insert(i - 1, i);  
    }  
    while (m--) {  
        int a, x, y;  
        cin >> a;  
        if (a == 1) {  
            cin >> x >> y;  
            erase(x);  
            insertLeft(y, x);  
        } else if (a == 2) {  
            cin >> x >> y;  
            erase(x);  
            insert(y, x);  
        } else {  
            cin >> x;  
            erase(x);  
        }  
    }  
    int i = arr[0].next;  
    if (i == 0) {  
        cout << "Empty!";  
        return;  
    }  
    while (i != 0) {  
        cout << arr[i].val << " ";  
        i = arr[i].next;  
    }  
}  
  
int main() {  
    ios::sync_with_stdio(false);  
    cin.tie(NULL);  
    cout.tie(NULL);  
    solve();  
    return 0;  
}
```

[21. 合并两个有序链表 - 力扣（LeetCode）](https://leetcode.cn/problems/merge-two-sorted-lists/)
```c++
class Solution {
public:
    ListNode *mergeTwoLists(ListNode *list1, ListNode *list2) {
        if (list1 == nullptr)
            return list2;
        if (list2 == nullptr)
            return list1;
        ListNode *head, *now;
        if (list1->val < list2->val) {
            head = list1;
            list1 = list1->next;
        } else {
            head = list2;
            list2 = list2->next;
        }
        now = head;
        while (list1 != nullptr && list2 != nullptr) {
            if (list1->val < list2->val) {
                now->next = list1;
                list1 = list1->next;
            } else {
                now->next = list2;
                list2 = list2->next;
            }
            now = now->next;
        }
        if (list1 != nullptr)
            now->next = list1;
        else
            now->next = list2;
        return head;
    }
};

```

[86. 分隔链表 - 力扣（LeetCode）](https://leetcode.cn/problems/partition-list/description/)
```c++
class Solution {
public:
    ListNode *partition(ListNode *head, int x) {
        ListNode *leftTail = nullptr, *leftHead = nullptr;
        ListNode *rightTail = nullptr, *rightHead = nullptr;
        ListNode *next = nullptr;
        while (head != nullptr) {
            next = head->next;
            head->next = nullptr;
            if (head->val < x) {
                if (leftHead == nullptr)
                    leftHead = head;
                else
                    leftTail->next = head;
                leftTail = head;
            } else {
                if (rightHead == nullptr)
                    rightHead = head;
                else
                    rightTail->next = head;
                rightTail = head;
            }
            head = next;
        }
        if (leftHead == nullptr)
            return rightHead;
        leftTail->next = rightHead;
        return leftHead;
    }
};
```
