#### 在有序数组中确定num存在还是不存在:
在一段有序的数组中，每次求剩余数的中点，看这个中点的数是比要找的数更小还是更大，若更小则表明中点左边所有数都比要找的数小，因此往右边找，若更大则表明中点右边所有数都比要找的数大，因此往左边找。这样每次砍掉剩余的一半，到最后就只剩一个数。若这个数是要找的数则存在，不是就不存在

1,3,5,9,13,15,23,25 想知道13是否存在？
$(0+7)/2=3$        v\[3\]=9         9<13        往右找
$(4+7)/2=5$        v\[5\]=15       15>13      往左找
4                          v\[4\]=13       存在

```c++
bool exist(vector<int> &arr, int num) {
	//如果arr长度为0，直接返回false
	if (arr.size() == 0) {
		return false;
	}

    // 初始化查找区间为整个数组，左闭右闭 [l, r]
    int l = 0;                      // 左指针，指向区间第一个元素
    int r = arr.size() - 1;           // 右指针，指向区间最后一个元素

    // 当左指针不超过右指针时，区间非空，继续查找
    while (l <= r) {
        // 计算中间位置下标（注意：这种写法在 l+r 很大时可能整数溢出，
        int mid = l + (r - l) / 2;

        // 情况1：中间元素正好等于目标值，查找成功
        if (arr[mid] == num)
            return true;

        // 情况2：中间元素小于目标值，说明目标值只可能在右半部分
        // 因为数组有序，左半部分所有元素都 <= v[mid] < num
        if (arr[mid] < num)
            l = mid + 1;   // 将左指针移到 mid 右侧，缩小查找范围
        // 情况3：中间元素大于目标值，说明目标值只可能在左半部分
        else
            r = mid - 1;   // 将右指针移到 mid 左侧，缩小查找范围
    }

    // 循环结束说明区间为空（l > r），仍未找到目标值
    return false;
}
```

#### 在有序数组中找>=num的最左位置
在一段有序的数组中，每次求剩余数的中点，看这个中点的数是比要找的数更小还是大于等于，若更小则表明中点左边所有数都比要找的数小，因此往右边找，若大于等于往则表明中点右边所有数都比要找的数大于等于，这个中点有可能是答案,因此更新答案往左边找。

3，6，6，7，9，13，15，19   找>=15

```c++
// 在升序数组 arr 中查找第一个大于等于 nums 的元素下标
// 如果不存在这样的元素，则返回 -1
int findLeft(vector<int> &arr, int nums) {
    int l = 0, r = arr.size() - 1; // 初始化二分查找区间为 [0, n-1]
    int ans = -1;                  // 记录当前找到的候选下标，初始为 -1 表示未找到

    while (l <= r) {               // 当查找区间非空时继续
        int m = l + (r - l) / 2;       // 取区间中间位置

        if (arr[m] >= nums) {      // 中间元素已经满足条件
            ans = m;               // 先记录当前下标
            r = m - 1;             // 继续在左半区查找更靠左的满足条件位置
        } else {                   // 中间元素小于目标值
            l = m + 1;             // 只能在右半区查找
        }
    }

    return ans;                    // 返回第一个满足 arr[ans] >= nums 的下标，未找到则为 -1
}
```

#### 在有序数组中找>=num的最左位置
在一段有序的数组中，每次求剩余数的中点，看这个中点的数是比要找的数小于等于还是大于，若小于等于往则表明中点左边所有数都比要找的数小，这个中点有可能是答案,因此更新答案往右边找，若大于往则表明中点右边所有数都比要找的数大于,因此往左边找。
```c++
// 在升序数组 arr 中查找最后一个小于等于 nums 的元素下标
// 如果不存在这样的元素，则返回 -1
int findRight(vector<int> &arr, int nums) {
    int l = 0, r = arr.size() - 1; // 初始化二分查找区间为 [0, n-1]
    int ans = -1;                  // 记录当前找到的候选下标，初始为 -1 表示未找到

    while (l <= r) {               // 当查找区间非空时继续
        int m = (l + r) / 2;       // 取区间中间位置

        if (arr[m] <= nums) {      // 中间元素满足条件（小于等于目标值）
            ans = m;               // 先记录当前下标
            l = m + 1;             // 继续在右半区查找更靠右的满足条件位置
        } else {                   // 中间元素大于目标值
            r = m - 1;             // 只能在左半区查找
        }
    }

    return ans;                    // 返回最后一个满足 arr[ans] <= nums 的下标，未找到则为 -1
}
```

#### 二分搜索不一定发生在有序数组上
[162. 寻找峰值 - 力扣（LeetCode）](https://leetcode.cn/problems/find-peak-element/)
先判断0，和n-1是不是峰值，是就直接输出，不是就表明1到n-2之间一定有一个峰值，取一个中点，若这个中点是一个峰值就直接输出，不是则表明两边一定有一个比这个中点大，那那个方向一定有一个峰值。
```c++
class Solution {
public:
    int exist(vector<int> &arr) {
        int l=0,r=arr.size()-1;
        while(l<=r) {
            int m=l+(r-l)/2;
            if (arr[m]>arr[m-1]&&arr[m]>arr[m+1])
                return m;
            if (arr[m-1]>arr[m])
                r=m-1;
            else
                l=m+1;
        }
    }

    int findPeakElement(vector<int>& nums) {
        int n=nums.size()-1;
        if (nums.size()==1 || nums[0]>nums[n-1])
            return 0;
        if (nums[1]>nums[0])
            return 1;
        return exist(nums);
    }
};
```
