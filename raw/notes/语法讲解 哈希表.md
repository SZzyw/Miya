c++：哈希表

将一个很大的空间对应到一个很小的空间下。
这就会造成冲突，比如两个3，对应的地址是一样的。
因此我们要解决哈希冲突。

#### 开放寻址法
像找一个坑位上厕所，到指定坑位时发现有人，就去下一个坑位找找
举例：哈希表大小为，6，分别插入：4，4，5

```c++
struct HashTable {
    vector<int> arr;
    vector<bool> vis;

    HashTable(int n) {
        arr.assign(n, 0);
        vis.assign(n, false);
    }

    int find(int x) {
        int n = arr.size();
        int t = (x % n + n) % n;
        while (arr[t] != x && vis[t] == true) {
            t = (t + 1) % n;
        }
        return t;
    }

    void insert(int x) {
        int pos = find(x);
        if (vis[pos] == false) {
            arr[pos] = x;
            vis[pos] = true;
        }
    }
};
```

```c++
#include <bits/stdc++.h>
using namespace std;

struct Hash_table {
    vector<int> arr;
    vector<bool> vis;

    Hash_table(int n) {
        arr.assign(n, 0);
        vis.assign(n, false);
    }

    int find(int x) {
        int n = arr.size();
        int t = (x % n + n) % n;
        while (arr[t] != x && vis[t] == true) {
            t = (t + 1) % n;
        }
        return t;
    }

    bool insert(int x) {
        int pos = find(x);
        if (vis[pos] == false) {
            arr[pos] = x;
            vis[pos] = true;
            return true;
        }
        return false;
    }
};

void solve() {
    int n;
    cin >> n;
    Hash_table t(n);
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        if (t.insert(x))
            cout << x << ' ';
    }
    cout << endl;
}

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    int T;
    cin >> T;
    while (T--)
        solve();
    return 0;
}
```

#### 拉链法
像找一个坑位上厕所，到指定坑位时发现有人，就去下一个坑位找找。
通过链表的方式
举例：4，4，5
```c++
struct Node {
    int val;
    int next;

    Node(int val = -1, int next = -1) : val(val), next(next) {
    }
};

struct ChainedHashTable {
    vector<Node> arr, head;

    ChainedHashTable(int n) : head(n) {
        head.assign(n, Node());
        arr.clear();
    }

    int find(int x) {
        int n = head.size();
        int t = (x % n + n) % n;
        for (int i = head[t].next; i != -1; i = arr[i].next) {
            int val = arr[i].val;
            if (val == x)
                return x;
        }
        return -1;
    }

    void insert(int x) {
        int pos = find(x);
        int n = head.size();
        int t = (x % n + n) % n;
        if (pos == -1) {
            arr.push_back(Node(x, head[t].next));
            head[t].next = arr.size() - 1;
        }
    }
};
```
