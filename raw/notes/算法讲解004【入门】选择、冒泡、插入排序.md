数据结构：排序
#### 选择
选择排序一句话：i~n-1范围上，找到最小值并放在i位置，然后i+1~n-1范围上继续。
```c++
void selectionSort(vector<int> &v) {
    int n=v.size();
    for (int i = 0; i < n; ++i) { // i 表示已排序部分的末尾下标
        int min_j = i; // 假设当前位置 i 是最小值
        for (int j = i + 1; j < n; ++j) // 在未排序部分（i+1 ~ n-1）中寻找更小值
            if (v[j] < v[min_j])
                min_j = j; // 更新最小值下标
        swap(v[i], v[min_j]); // 将最小值交换到已排序部分的末尾
    }
}
```

#### 冒泡
冒泡排序一句话：0~i范围上，相邻位置较大的数滚下去，最大值最终来到i位置，然后0~i-1范围上继续
```c++
void bubbleSort(vector<int> &v) {
    int n=v.size();
    for (int i = n - 1; i >= 0; --i) // i 表示未排序部分的末尾下标，从后往前
        for (int j = 0; j < i; ++j) // 相邻比较，范围是 [0, i-1]
            if (v[j] > v[j + 1]) // 如果前一个比后一个大，交换
                swap(v[j], v[j + 1]);
}
```

#### 插入排序
插入排序一句话：0~i范围上已经有序，新来的数从右到左滑到不再小的位置插入，然后继续
```c++
void insertionSort(vector<int> &v) {
    int n=v.size();
    for (int i = 1; i < n; ++i) { // i 表示待插入元素的初始下标，从第二个开始
        int value = v[i]; // 保存当前待插入的值
        int j = i - 1; // j 指向已排序部分的末尾
        // 将比 value 大的元素依次后移一位，为 value 腾出位置
        while (j >= 0 && v[j] > value) {
            v[j + 1] = v[j]; // 后移
            --j;
        }
        v[j + 1] = value; // 将 value 插入到正确的位置
    }
}
```

#### 例题
[P1177 【模板】排序 - 洛谷](https://www.luogu.com.cn/problem/P1177)
``` c++
#include <bits/stdc++.h>
using namespace std;
void func(vector<int> &v) {
    int n=v.size();
    for (int i = n - 1; i >= 0; --i) // i 表示未排序部分的末尾下标，从后往前
        for (int j = 0; j < i; ++j) // 相邻比较，范围是 [0, i-1]
            if (v[j] > v[j + 1]) // 如果前一个比后一个大，交换
                swap(v[j], v[j + 1]);
}

void solve() {
    int n;
    cin >> n; // 读取整数个数
    vector<int> v(n); // 创建动态数组存储 n 个整数
    for (int i = 0; i < n; ++i) {
        cin >> v[i]; // 依次读入每个数
    }

    func(v);

    // 输出排序后的结果（若所有排序块都被屏蔽，则输出原始顺序）
    for (auto it: v) { // 范围 for 循环，依次取出 v 中的每个元素
        cout << it << " "; // 输出元素，以空格分隔
    }
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    solve();
    return 0;
}
```
