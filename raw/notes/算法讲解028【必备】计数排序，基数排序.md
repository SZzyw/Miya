数据结构：排序
#### 计数排序
统计每个数字出现的个数，然后遍历就行了。
举例:5,2,5,5,3,2.
2有2次，3有1次，5有3次。因此是2，2，3，5，5，5

#### 基数排序
不能排负数
如果有负数，先把数组减去最小数字，再去排序，排完后再加回来。
思路：先按个位数用从小到大排序，再按十位数从小到大排序，再按百位数从小到大排序。
原理：队列是先进先出，之前排好的相对顺序不会变。
先10个队列，表明个位0~9，遍历数组，放到对应的队列中。遍历完后再依次拿出来
在10个队列，表明十位0~9，遍历数组，放到对应的队列中。遍历完后再依次拿出来
....
举例：23，51，42，24，6

```c++
struct RadixSort {
    queue<int> s[10];

    void radixSort(vector<int> &v) {
        int mmax = 0;
        for (auto it: v)
            mmax = max(it, mmax);
        for (int i = 1; mmax / i > 0; i *= 10) {
            for (auto it: v) {
                int value = it / i % 10;
                s[value].push(it);
            }
            v.clear();
            for (int i = 0; i < 10; i++) {
                while (s[i].size()) {
                    v.push_back(s[i].front());
                    s[i].pop();
                }
            }
        }
    }
};
```

#### 模板题
[P1177 【模板】排序 - 洛谷](https://www.luogu.com.cn/problem/P1177)
```c++
#include <bits/stdc++.h>
using namespace std;

struct RadixSort {
    queue<int> s[10];

    void radixSort(vector<int> &v) {
        int mmax = 0;
        for (auto it: v)
            mmax = max(it, mmax);
        for (int i = 1; mmax / i > 0; i *= 10) {
            for (auto it: v) {
                int value = it / i % 10;
                s[value].push(it);
            }
            v.clear();
            for (int i = 0; i < 10; i++) {
                while (s[i].size()) {
                    v.push_back(s[i].front());
                    s[i].pop();
                }
            }
        }
    }
};

void solve() {
    RadixSort ridix_sort;
    int n;
    cin >> n;
    vector<int> v;
    v.assign(n, 0);
    for (int i = 0; i < n; i++) {
        cin >> v[i];
    }
    ridix_sort.radixSort(v);
    for (auto it: v) {
        cout << it << " ";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    solve();
    return 0;
}
```
