#### 快速排序
从数组中随机选择一个数，按这个数对数组进行排序，小于等于放左边，大于放右边，按这个数分成左右两段，继续排序。
举例：5，3，6，7，3
```c++
struct QuickSort {
    vector<int> &arr;

    QuickSort(vector<int> &arr) : arr(arr) {
    }

    int partition(int l, int r, int x) {
        int j = l, now = -1;
        for (int i = l; i <= r; i++) {
            if (arr[i] <= x) {
                swap(arr[i], arr[j]);
                if (arr[j] == x)
                    now = j;
                j++;
            }
        }
        j--;
        swap(arr[now], arr[j]);
        return j;
    }

    void quickSort(int l, int r) {
        if (l >= r)
            return;
        int x = arr[rand() % (r - l + 1) + l];
        int mid = partition(l, r, x);
        quickSort(l, mid - 1);
        quickSort(mid + 1, r);
    }
};
```

#### 优化
原本每次只能确定一个数的正确位置，我们可以把想等的放在一段。也就是小于左边，等于中间，大于右边。我们只需要把小于的往左边移，大于的往右边移动
举例：1，2，2，1，3
```
struct QuickSort {
    vector<int> &arr;

    QuickSort(vector<int> &arr) : arr(arr) {
    }

    array<int, 2> partition(int l, int r, int x) {
        int first = l, last = r;
        for (int i = l; i <= last; i++) {
            if (arr[i] < x)
                swap(arr[i], arr[first++]);
            else if (arr[i] > x)
                swap(arr[i--], arr[last--]); //注意要i--，换过来的数还没有监测
        }
        return {first, last};
    }

    void quickSort(int l, int r) {
        if (l >= r)
            return;
        int x = arr[rand() % (r - l + 1) + l];
        array<int, 2> mid = partition(l, r, x);
        quickSort(l, mid[0] - 1);
        quickSort(mid[1] + 1, r);
    }
};
```

#### 模板题

```c++
#include <bits/stdc++.h>

struct QuickSort {
    vector<int> &arr;

    QuickSort(vector<int> &arr) : arr(arr) {
    }

        int first = l, last = r;
        for (int i = l; i <= last; i++) {
            if (arr[i] < x)
                swap(arr[i], arr[first++]);
            else if (arr[i] > x)
                swap(arr[i--], arr[last--]); //注意要i--，换过来的数还没有监测
        }
        return {first, last};
    }

    void quickSort(int l, int r) {
        if (l >= r)
            return;
        int x = arr[rand() % (r - l + 1) + l];
        array<int, 2> mid = partition(l, r, x);
        quickSort(l, mid[0] - 1);
        quickSort(mid[1] + 1, r);
    }
};

vector<int> nums;

void solve() {
    int n;
    cin >> n;
    nums.resize(n);
    for (int i = 0; i < n; i++)
        cin >> nums[i];
    QuickSort quick_sort(nums);
    quick_sort.quickSort(0, nums.size() - 1);
    for (auto it: nums) {
        cout << it << " ";
    }
}
```

#### 时间复杂度和空间复杂度
随机数算复杂度不能按最坏情况算
最坏时间复杂度是O($n^2$)   举例\[1,2,3,4,5,6\]
最优时间复杂度是O(n)     举例\[1,1,1,1,1,1\]

理想状态，每次都分一半，那就和归并排序一样，有logn层，每层n个。

随机数的行为既不能直接最好也不能直接最坏，通过期望，正态分布来进行计算，时间复杂度是$nlog n$，空间复杂度是$logn$.,
