数据结构：分治
#### 归并分治
1.思考一个问题在大范围上的答案，是否等于，左部分的答案 + 右部分的答案 + 跨越左右产生的答案
2.计算“跨越左右产生的答案”时，如果加上左、右各自有序这个设定，会不会获得计算的便利性
如果以上两点都成立，那么该问题很可能被归并分治解决
求解答案的过程中只需要加入归并排序的过程即可，因为要让左、右各自有序，来获得计算的便利性
举例：5 4 2 6 3 1 求小于等于v\[i\]的和

#### 模板题
[计算数组的小和_牛客题霸_牛客网](https://www.nowcoder.com/practice/edfe05a1d45c4ea89101d936cac32469)
```c++
#include <bits/stdc++.h>
using namespace std;
#define int long long

struct SmallSum {
    vector<int> &arr, help;

    SmallSum(vector<int> &arr) : arr(arr) {
        help.resize(arr.size());
    }

    int merge(int l, int m, int r) {
        int ans = 0, sum = 0;
        for (int k = l, i = l, j = m + 1; k <= r; k++) {
            if (j == r + 1 || (i <= m && arr[i] <= arr[j])) {
                sum += arr[i];
                help[k] = arr[i];
                i++;
            } else {
                ans += sum;
                help[k] = arr[j];
                j++;
            }
        }
        for (int i = l; i <= r; i++) {
            arr[i] = help[i];
        }
        return ans;
    }

    int smallSum(int l, int r) {
        if (l == r)
            return 0;
        int m = l + (r - l) / 2;
        return smallSum(l, m) + smallSum(m + 1, r) + merge(l, m, r);
    }
};

vector<int> nums;

void solve() {
    int n;
    cin >> n;
    nums.resize(n);
    for (int i = 0; i < n; i++)
        cin >> nums[i];
    SmallSum small_sum(nums);
    cout << small_sum.smallSum(0, nums.size() - 1);
}

signed main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    int T = 1;
    // cin>>T;
    while (T--)
        solve();
    return 0;
}

```

#### 例题
[P1908 逆序对 - 洛谷](https://www.luogu.com.cn/problem/P1908)
```c++
#include <bits/stdc++.h>
using namespace std;
#define int long long

struct ReversePairs {
    vector<int> &arr, help;

    ReversePairs(vector<int> &arr) : arr(arr) {
        help.resize(arr.size());
    }

    int merge(int l, int m, int r) {
        int ans = 0, sum = 0;
        for (int k = l, i = l, j = m + 1; k <= r; k++) {
            if (j == r + 1 || (i <= m && arr[i] <= arr[j])) {
                ans += sum;
                help[k] = arr[i];
                i++;
            } else {
                sum++;
                help[k] = arr[j];
                j++;
            }
        }
        for (int i = l; i <= r; i++) {
            arr[i] = help[i];
        }
        return ans;
    }

    int reversePairs(int l, int r) {
        if (l == r)
            return 0;
        int m = l + (r - l) / 2;
        return reversePairs(l, m) + reversePairs(m + 1, r) + merge(l, m, r);
    }
};

vector<int> nums;

void solve() {
    int n;
    cin >> n;
    nums.resize(n);
    for (int i = 0; i < n; i++)
        cin >> nums[i];
    ReversePairs reverse_pairs(nums);
    cout << reverse_pairs.reversePairs(0, nums.size() - 1);
}

signed main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    int T = 1;
    // cin>>T;
    while (T--)
        solve();
    return 0;
}
```
