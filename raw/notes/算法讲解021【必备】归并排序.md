#### 归并排序
类似讲递归时最大值的思路。
左边排好序，右边排好序，在合并的时候保持整体有序
merge过程：谁小拷贝谁，直到左右两部分所有的数字耗尽，拷贝回原数组
举例：6，4，2，3，9，4

```c++
struct MergeSort {
    vector<int> &arr, help;

    MergeSort(vector<int> &arr) : arr(arr) {
        help.resize(arr.size());
    }

    void merge(int l, int m, int r) {
        for (int k = l, i = l, j = m + 1; k <= r; k++) {
            if (j == r + 1 || (i <= m && arr[i] <= arr[j])) {
                help[k] = arr[i];
                i++;
            } else {
                help[k] = arr[j];
                j++;
            }
        }
        for (int i = l; i <= r; i++) {
            arr[i] = help[i];
        }
    }

    void mergeSort(int l, int r) {
        if (l == r)
            return;
        int m = l + (r - l) / 2;
        mergeSort(l, m);
        mergeSort(m + 1, r);
        merge(l, m, r);
    }
};
```
非递归版可以通过步长来完成，步长：1，2，4.....

#### 模板题
[P1177 【模板】排序 - 洛谷](https://www.luogu.com.cn/problem/P1177)
```c++
#include <bits/stdc++.h>
using namespace std;

struct MergeSort {
    vector<int> &arr, help;

    MergeSort(vector<int> &arr) : arr(arr) {
        help.resize(arr.size());
    }

    void merge(int l, int m, int r) {
        for (int k = l, i = l, j = m + 1; k <= r; k++) {
            if (j == r + 1 || (i <= m && arr[i] <= arr[j])) {
                help[k] = arr[i];
                i++;
            } else {
                help[k] = arr[j];
                j++;
            }
        }
        for (int i = l; i <= r; i++) {
            arr[i] = help[i];
        }
    }

    void mergeSort(int l, int r) {
        if (l == r)
            return;
        int m = l + (r - l) / 2;
        mergeSort(l, m);
        mergeSort(m + 1, r);
        merge(l, m, r);
    }
};

vector<int> nums;

void solve() {
    int n;
    cin >> n;
    nums.resize(n);
    for (int i = 0; i < n; i++)
        cin >> nums[i];
    MergeSort sor(nums);
    sor.mergeSort(0, nums.size() - 1);
    for (auto it: nums) {
        cout << it << " ";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    int T = 1;
    // cin>>T;
    while (T--)
        solve();
    return 0;
}

```

#### 时间复杂度和空间复杂度。
每次都要分一半，因此$log n$层，每层都要把数组遍历一遍，因此n。总复杂度：O($nlogn$)
最多递归$logn$，空间复杂度为O($logn$)

#### 其他
为什么归并排序比别的排序快呢，因为归并排序比较行为没有浪费。

有些资料说可以用原地归并排序，把额外空间复杂度变成O(1)，不要浪费时间去学。因为原地归并排序确实可以省空间，但是会让复杂度变成O(n^2)
