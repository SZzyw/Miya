#### 完全二叉树
完全二叉树和数组前缀范围来对应，大小，单独的变量size来控制。i的父亲节点：$(i-1)/2$，i的左孩子：$i*2 + 1$，i的右孩子：$i*2 + 2$
举例：\[3,5,3,6,2\]

#### 堆（大根堆，小根堆）
讲解只讲大根堆，小根堆同理
在完全二叉树的基础上，始终保持该节点比其子节点大与等于,此为大根堆，若不符合则需要进行操作。
##### 向上调整
对于节点i，若arr\[i\],若其比父节点大，则进行交换，并继续在新节点的位置与其父节点比较。
举例：\[5,6,3,6\]

```c++
    void heapInsert(vector<int> &arr, int i) {
        while (arr[i] > arr[(i - 1) / 2]) { //-1/2为0
            swap(arr[i], arr[(i - 1) / 2]);
            i = (i - 1) / 2;
        }
    }
```

#### 向下调整
对于节点i，若arr\[i\]比其子节点更小，则进行交换，并继续在新节点的位置与子父节点比较。注意是跟子结点中更大的哪个比
举例：\[6,6,5,3\]，arr\[0\]变为2.

```c++
    void heappify(vector<int> &arr, int i, int siz) {
        while ((i * 2 + 1) < siz) {
            int l = i * 2 + 1, r = l + 1;
            int best = l;
            if (r < siz && arr[l] < arr[r])
                best = r;
            if (arr[i] < arr[best])
                swap(arr[i], arr[best]);
            else
                break;
            i = best;
        }
    }
```

#### 时间复杂度
无论是向上调整还是向下调整，每次调整的最多的数为树的高度，而数组长度为n时，树的高度为$log n$ 
因此，时间复杂度为$log n$ 

#### 模板题
[P3378 【模板】堆 - 洛谷](https://www.luogu.com.cn/problem/P3378)

``` c++
#include <bits/stdc++.h>
using namespace std;

struct Heap {
    void heapInsert(vector<int> &arr, int i) {
        while (arr[i] < arr[(i - 1) / 2]) {
            //-1/2为0
            swap(arr[i], arr[(i - 1) / 2]);
            i = (i - 1) / 2;
        }
    }

    void heappify(vector<int> &arr, int i, int siz) {
        while ((i * 2 + 1) < siz) {
            int l = i * 2 + 1, r = l + 1;
            int best = l;
            if (r < siz && arr[l] > arr[r])
                best = r;
            if (arr[i] > arr[best])
                swap(arr[i], arr[best]);
            else
                break;
            i = best;
        }
    }
};

void solve() {
    int n;
    cin >> n;
    vector<int> arr;
    Heap heap;
    while (n--) {
        int op, x;
        cin >> op;
        if (op == 1) {
            cin >> x;
            arr.push_back(x);
            heap.heapInsert(arr, arr.size() - 1);
        } else if (op == 2) {
            cout << arr[0] << endl;
        } else {
            swap(arr.back(), arr[0]);
            arr.pop_back();
            heap.heappify(arr, 0, arr.size());
        }
    }
}

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    solve();
    return 0;
}
```

#### 例题
[合并k个已排序的链表_牛客题霸_牛客网](https://www.nowcoder.com/practice/65cfde9e5b9b4cf2b6bafa5f3ef33fa6)
先把所有链表的头节点放入堆中，取出堆最小的节点，再把该节点的下一个放入堆中

```c++
struct Compare {
    bool operator()(const ListNode *a, const ListNode *b) {
        return a->val > b->val;
    }
};

class Solution {
public:
    ListNode *mergeKLists(vector<ListNode *> &lists) {
        priority_queue<ListNode *, vector<ListNode *>, Compare> pq;
        for (auto it: lists) {
            if (it != nullptr)
                pq.push(it);
        }
        ListNode *ans = nullptr, *head = nullptr;
        while (!pq.empty()) {
            ListNode *tmp = pq.top();
            pq.pop();
            if (ans == nullptr) {
                ans = tmp;
                head = tmp;
            } else {
                head->next = tmp;
                head = tmp;
            }
            if (tmp->next != nullptr)
                pq.push(tmp->next);
            head->next =
                    nullptr; //注意：务必需要把tmp->next存进去了再把head->next设为空，此时head==tmp
        }
        return ans;
    }
};

```

时间复杂度：$nlogn$


[线段重合_牛客题霸_牛客网](https://www.nowcoder.com/practice/1ae8d0b6bb4e4bcdbf64ec491f63fc37)
任何一个区域的左边界，都一定是某一条线段的左边界
因此我们讨论有多少个重合区域以当前选择的区域的左边界为左边界
因此我们按l从小到大排序，因此我们遍历到该区域的l时，该区域的l一定比之前遍历过的l大，从而满足了上一个条件
我们再把之前遍历过的r放在小根堆里，如果说小根堆里最小的r比l小与等于，则说明没有相交。如果大于的，说明该小根堆里剩余的所有区间都与该区间相交。
再取max就是最大的数量，又因为是线段的数量所有答案+1

```c++
#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    cin >> n;
    priority_queue<int, vector<int>, greater<int> > pq;
    vector<array<int, 2> > arr;
    arr.assign(n, {});
    for (int i = 0; i < n; i++) {
        cin >> arr[i][0] >> arr[i][1];
    }
    sort(arr.begin(), arr.end());
    int mmax = 0;
    for (int i = 0; i < n; i++) {
        while (pq.size() && pq.top() <= arr[i][0])
            pq.pop();
        mmax = max(mmax, (int) pq.size());
        pq.push(arr[i][1]);
    }
    cout << mmax + 1 << endl;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    solve();
    return 0;
}
```

时间复杂度：$nlogn$
