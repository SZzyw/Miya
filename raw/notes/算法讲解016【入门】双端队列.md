[641. 设计循环双端队列 - 力扣（LeetCode）](https://leetcode.cn/problems/design-circular-deque/description/)

```c++
class MyCircularDeque {
public:
    vector<int> deq;
    int l, r, n;

    MyCircularDeque(int k) {
        deq.resize(k + 10);
        n = k + 10;
        l = 0, r = -1;
    }

    int size() {
        return r - l + 1;
    }

    bool insertFront(int value) {
        if (this->size() == n)
            return false;
        l--;
        deq[(l + n) % n] = value;
        return true;
    }

    bool insertLast(int value) {
        if (this->size() == n)
            return false;
        r++;
        deq[(r + n) % n] = value;
        return true;
    }

    bool deleteFront() {
        if (this->size() == 0)
            return false;
        l++;
        return true;
    }

    bool deleteLast() {
        if (this->size() == 0)
            return false;
        r--;
        return true;
    }

    int getFront() {
        if (this->size() == 0)
            return -1;
        return deq[(l + n) % n];
    }

    int getRear() {
        if (this->size() == 0)
            return -1;
        return deq[(r + n) % n];
    }

    bool isEmpty() {
        if (this->size() == 0)
            return true;
        return false;
    }

    bool isFull() {
        if (this->size() == 0)
            return false;
        return true;
    }
};
```
