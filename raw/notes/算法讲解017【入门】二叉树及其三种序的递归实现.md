#### 什么是二叉树
每个节点最多有两个子节点的图是二叉树
由于每个点最多两个子节点，我们可以每个点都是arrat<int,2>类型的，\[0\]就是左子树的节点，\[1\]就是右子树的节点

```c++
void preOrder(int head) {
```

栈
由于每次中会最先执行，因此直接输出中。由于栈的特性，先推入右节点，再推入左节点。
```c++
void preOrder(int head) {  
    stack<int> s;  
    s.push(head);  
    while (s.size()) {  
        head = s.top();  
        s.pop();  
        cout << head << " ";  
        if (tree[head][1] != 0)  
            s.push(tree[head][1]);  
        if (tree[head][0] != 0)  
            s.push(tree[head][0]);  
    }  
}
```

#### 中序遍历
左->中->右
举例：7个点的满二叉树
```c++
void inOrder(int head) {
    if (head == 0)
        return;
    inOrder(tree[head][0]);
    cout << head << ' ';
    inOrder(tree[head][1]);
}
```

栈
类似与递归的思路，先去左边，左边弄完后，回来直接输出(弹数来一个),下一个top就是右边的了。
1.子树头，左边界进栈
2.栈弹出节点打印，节点右树进栈，重1
3.没子树且栈空
```c++
void inOrder(int head) {
    stack<int> s;
    s.push(head);
    while (s.size()) {
        head = s.top();
        s.pop();
        if (head == 0) {
            if (s.size()) {
                cout << s.top() << ' ';
                s.pop();
            }
            continue;
        }
        s.push(tree[head][1]);
        s.push(head);
        s.push(tree[head][0]);
    }
}
```

#### 后序遍历
左->右->中
举例：7个点的满二叉树
递归
```c++
void posOrder(int head) {
    if (head == 0)
        return;
    posOrder(tree[head][0]);
    posOrder(tree[head][1]);
    cout << head << ' ';
}
```

栈
上次打印的节点一定是某个节点左子树节点或右子树节点。 如果上次打印的是左子树节点，那就去右子树的节点。如果上次打印的是右子树节点，那就打印自己节点。
```c++
void posOrder(int head) {  
    stack<int> s;  
    s.push(head);  
    while (s.size()) {  
        int cur = s.top();  
        if (tree[cur][0] != 0 && head != tree[cur][0] && head != tree[cur][1])  
            s.push(tree[cur][0]);  
        else if (tree[cur][1] != 0 && head != tree[cur][1])  
            s.push(tree[cur][1]);  
        else {  
            cout << cur << ' ';  
            head = cur;  
            s.pop();  
        }  
    }  
}
```
#### 时间复杂度和空间复杂度
三种遍历的时间复杂度均为O(n)
空间复杂度为树的高度。

#### 模板题
[B3642 二叉树的遍历 - 洛谷](https://www.luogu.com.cn/problem/B3642)

```c++
#include <bits/stdc++.h>
using namespace std;
using PII = array<int, 2>;
vector<PII> tree;

void preOrder(int head) {
    if (head == 0)
        return;
    cout << head << ' ';
    preOrder(tree[head][0]);
    preOrder(tree[head][1]);
}

void inOrder(int head) {
    if (head == 0)
        return;
    inOrder(tree[head][0]);
    cout << head << ' ';
    inOrder(tree[head][1]);
}

void posOrder(int head) {
    if (head == 0)
        return;
    posOrder(tree[head][0]);
    posOrder(tree[head][1]);
    cout << head << ' ';
}

void solve() {
    int n;
    cin >> n;
    tree.resize(n + 1);
    for (int i = 1; i <= n; i++) {
        int x, y;
        cin >> x >> y;
        tree[i] = {x, y};
    }
    preOrder(1);
    cout << endl;
    inOrder(1);
    cout << endl;

    cout << endl;

int main() {
    ios::sync_with_stdio(false);

    // cin>>T;
```

#### 例题
[P1030 [NOIP 2001 普及组] 求先序排列 - 洛谷](https://www.luogu.com.cn/problem/P1030)
```c++
#include <bits/stdc++.h>
using namespace std;
using PII = array<int, 2>;
vector<PII> tree(20);

int preFind(string in, string pos) {
    int n = in.size();
    if (n == 0)
        return -1;
    int num = pos[n - 1] - 'A';
    int k = in.find(pos[n - 1]);
    pos.erase(pos.end() - 1);
    tree[num][0] = preFind(in.substr(0, k), pos.substr(0, k));
    tree[num][1] = preFind(in.substr(k + 1), pos.substr(k));
    return num;
}

void preOrder(int head) {
    if (head == -1)
        return;
    cout << char(head + 'A');
    preOrder(tree[head][0]);
    preOrder(tree[head][1]);
}

void solve() {
    string in, pos;
    cin >> in >> pos;
    preOrder(preFind(in, pos));
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    int T = 1;
    // cin>>T;
    while (T--)
        solve();
    return 0;
}
```
