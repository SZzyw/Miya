数据结构：队列，栈
#### 队列
特点是先进先出，就像去食堂买饭排队。
举例：1进，5进，2进，出一个，出一个，3进
循环队列：尾部和头部连起来的队列，代码为循环队列

#### 队列模板题
[B3616 【模板】队列 - 洛谷](https://www.luogu.com.cn/problem/B3616)
```c++
#include <bits/stdc++.h>
using namespace std;

struct Queue {
    vector<int> queue;
    int l, r;

    Queue(int n) {
        queue.resize(n);
        l = 0, r = -1;
    }

    void push(int x) {
        queue[++r] = x;
    }

    void pop() {
        l++;
    }

    int size() {
        return r - l + 1;
    }

    int query() {
        return queue[l];
    }
};

void solve() {
    int m;
    cin >> m;
    Queue q(m + 10);
    while (m--) {
        int a, x;
        cin >> a;
        if (a == 1) {
            cin >> x;
            q.push(x);
        } else if (a == 2) {
            if (q.size() == 0)
                cout << "ERR_CANNOT_POP" << endl;
            else
                q.pop();
        } else if (a == 3) {
            if (q.size() == 0)
                cout << "ERR_CANNOT_QUERY" << endl;
            else
                cout << q.query() << endl;
        } else
            cout << q.size() << endl;
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    solve();
    return 0;
}

```

#### 栈
特点是先进后出，就像一头弹夹。
举例：1进，5进，2进，出一个，出一个，3进

#### 栈模板题
注意数据量
```c++
#include <bits/stdc++.h>
using namespace std;
#define int unsigned long long

struct Stack {
    vector<int> stack;

    void push(int x) {
        stack.push_back(x);
    }

    void pop() {
        stack.pop_back();
    }

    int query() {
        return stack.back();
    }

    int size() {
        return stack.size();
    }
};

void solve() {
    int m;
    cin >> m;
    Stack s;
    while (m--) {
        string a;
        int x;
        cin >> a;

                cout << "Empty" << endl;
            else
                s.pop();

            if (s.size() == 0)
                cout << "Anguei!" << endl;
            else
                cout << s.query() << endl;
        } else
            cout << s.size() << endl;
    }
}

signed main() {

    cin.tie(0);
    cout.tie(0);
```

#### 例题
[232. 用栈实现队列 - 力扣（LeetCode）](https://leetcode.cn/problems/implement-queue-using-stacks/description/)
```c++
class MyQueue {
public:
    stack<int> in, out;

    MyQueue() {
    }

    void push(int x) {
        in.push(x);
    }

    int pop() {
        if (out.size() == 0) {
            while (in.size()) {
                out.push(in.top());
                in.pop();
            }
        }
        int value = out.top();
        out.pop();
        return value;
    }

    int peek() {
        if (out.size() == 0) {
            while (in.size()) {
                out.push(in.top());
                in.pop();
            }
        }
        return out.top();
    }

    bool empty() {
        if (in.size() + out.size() > 0)
            return false;
        return true;
    }
};
```

[225. 用队列实现栈 - 力扣（LeetCode）](https://leetcode.cn/problems/implement-stack-using-queues/)
```c++
class MyStack {
public:
    queue<int> que;
    MyStack() {
        
    }
    
    void push(int x) {
        que.push(x);
    }
    
    int pop() {
        int n=que.size()-1;
        while(n--){
            que.push(que.front());
            que.pop();
        }
        int value=que.front();
        que.pop();
        return value;
    }
    
    int top() {
        int n=que.size()-1;
        while(n--){
            que.push(que.front());
            que.pop();
        }
        int value=que.front();
        que.push(value);//记得放回去
        que.pop();
        return value;
    }
    
    bool empty() {
        if(que.size()>0)
            return false;
        return true;
    }
};
```

[155. 最小栈 - 力扣（LeetCode）](https://leetcode.cn/problems/min-stack/)
```c++
class MinStack {
public:
    stack<int> sta, mmin;

    MinStack() {
        sta.push(INT_MAX);
        mmin.push(INT_MAX);
    }

    void push(int value) {
        sta.push(value);
        mmin.push(min(mmin.top(), value));
    }

    void pop() {
        sta.pop();
        mmin.pop();
    }

    int top() {
        return sta.top();
    }

    int getMin() {
        return mmin.top();
    }
};
```
